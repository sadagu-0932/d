section: 유형 14 함수의 극한의 활용
@ 53 | ④\ 2\sqrt2
\overline{AB}=\sqrt2\sqrt{1+4t},\ \frac{\sqrt2(\sqrt{1+4t}-1)}{t}=\frac{4\sqrt2}{\sqrt{1+4t}+1}\to2\sqrt2
@ 54 | \frac12
\overline{AB}=m,\ S=\frac{m^2}2\Rightarrow\frac{m^2}{2(m^2+1)}\to\frac12
@ 55 | ①\ \frac{\sqrt2}{2}
y_A=\sqrt{2t+2}-1,\ S=\frac t2(\sqrt{2t+2}-1)\Rightarrow\frac{S}{t\sqrt t}\to\frac{\sqrt2}2
@ 56 | ③\ \frac14
a^2+a=m^2,\ S=\frac{a\sqrt a}{2},\ \frac a{m^2}\to1\Rightarrow\frac{S^2}{m^6}\to\frac14
