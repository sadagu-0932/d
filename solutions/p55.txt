section: 04 도함수의 활용 (개념 확인 문제)
@ 01 | 0
y'=3x^2-4x+1\Rightarrow x=1:\ 3-4+1=0
@ 02 | 8
y'=6x^2-8x\Rightarrow x=2:\ 24-16=8
@ 03 | y=5x-11
y'=6x-7,\ x=2:\ 5\Rightarrow y-(-1)=5(x-2)
@ 04 | y=-7x-8
y'=-3x^2+4x,\ x=-1:\ -7\Rightarrow y+1=-7(x+1)
@ 05 | y=x+1
y'=4x^3-12x^2+1,\ x=0:\ 1\Rightarrow y-1=x
@ 06 | y=4x-1
y'=2x+2=4\Rightarrow x=1,\ y=3
y-3=4(x-1)
@ 07 | y=4x-7
y'=x+2=4\Rightarrow x=2,\ y=2+4-5=1
y-1=4(x-2)
@ 08 | y=4x+\frac13
x^2-2x+5=4\Rightarrow x=1,\ y=\frac{13}3:\ y=4(x-1)+\frac{13}3
@ 09 | y=4x-2
y'=2x=4\Rightarrow x=2,\ y=6\Rightarrow y-6=4(x-2)
@ 10 | y=-3x+17
3x+y-2=0\text{ 의 기울기 }-3:\ y'=-2x+5=-3\Rightarrow x=4,\ y=5
y-5=-3(x-4)
@ 11 | y=x,\ y=x+4
3x^2-2=1\Rightarrow x=\pm1:\ (1,1),\ (-1,3)\text{ 에서 기울기 }1
@ 12 | y=2x+3,\ y=-6x-5
(-1,1)\text{ 대입}:\ a^2+2a-3=0\Rightarrow a=1,-3
@ 13 | y=x-2,\ y=-7x+22
(3,1)\text{ 대입}:\ a^2-6a+5=0\Rightarrow a=1,5
@ 14 | y=x+2
(0,2)\text{ 대입}:\ -2a^3=2\Rightarrow a=-1,\ \text{기울기 }1
@ 15 | c=2
f(0)=f(4)=4,\quad f'(c)=4c-8=0\Rightarrow c=2
@ 16 | c=1
f(-2)=f(4)=5,\quad f'(c)=2c-2=0\Rightarrow c=1
@ 17 | c=0
f(-1)=f(2)=-1,\quad f'(c)=3c^2-6c=0\Rightarrow c=0\ (c=2\notin(-1,2))
@ 18 | c=\frac32
\frac{f(3)-f(0)}{3-0}=\frac{14-(-1)}3=5,\quad f'(c)=4c-1=5\Rightarrow c=\frac32
@ 19 | c=\frac72
\frac{f(5)-f(2)}{3}=\frac{28-4}3=8,\quad f'(c)=2c+1=8\Rightarrow c=\frac72
@ 20 | c=3
\frac{8-(-1)}3=3=3(c-2)^2\Rightarrow c=3
