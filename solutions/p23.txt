section: 02 함수의 연속 (개념 확인 문제)
@ 01 | \text{불연속}
f(1)\text{ 이 정의되지 않음 (빈 원)}\Rightarrow x=1\text{ 에서 불연속}
@ 02 | \text{불연속}
\lim_{x\to1-}f(x)=1\ne\lim_{x\to1+}f(x)=2\Rightarrow\lim_{x\to1}f(x)\text{ 없음}
@ 03 | \text{불연속}
\lim_{x\to1}f(x)=0\ne f(1)=2
@ 04 | \text{연속}
\lim_{x\to1}f(x)=1=f(1)
@ 05 | \text{불연속}
\lim_{x\to1-}(x+1)=2,\quad\lim_{x\to1+}(2x-1)=1\Rightarrow\text{극한 없음}
@ 06 | \text{연속}
\lim_{x\to1-}(x^2-1)=0,\quad\lim_{x\to1+}(2x-2)=0,\quad f(1)=0
@ 07 | \text{불연속}
\lim_{x\to1-}\frac1{x-1}=-\infty,\ \lim_{x\to1+}\frac1{x-1}=\infty\Rightarrow\text{극한 없음}
@ 08 | \text{연속}
\lim_{x\to1}\frac{(x-1)(x+1)}{x-1}=\lim_{x\to1}(x+1)=2=f(1)
@ 09 | \text{연속}
\lim_{x\to1-}(-x+1)=0,\quad\lim_{x\to1+}\sqrt{x-1}=0,\quad f(1)=0
@ 10 | (-\infty,\ \infty)
f(x)=2x^2+x-1\text{ 은 다항함수}\Rightarrow\text{모든 실수에서 연속}
@ 11 | (-\infty,-1)\cup(-1,\infty)
x+1\ne0\Rightarrow x\ne-1
@ 12 | [-2,\ \infty)
x+2\ge0\Rightarrow x\ge-2
@ 13 | (-\infty,\ \infty)
f(x)+g(x)=2x-1
@ 14 | (-\infty,\ \infty)
2f(x)-g(x)=2(x+1)-(x-2)=x+4
@ 15 | (-\infty,\ \infty)
f(x)g(x)=(x+1)(x-2)=x^2-x-2
@ 16 | (-\infty,2)\cup(2,\infty)
\frac{f(x)}{g(x)}=\frac{x+1}{x-2},\quad g(x)\ne0\Rightarrow x\ne2
@ 17 | \text{최댓값 없음, 최솟값 }-1
f(x)=(x-1)^2-1,\ \text{구간 }(-1,2)
x=1\text{ 에서 최솟값 }f(1)=-1
x\to-1+\text{ 이면 }f(x)\to3\text{ 이지만 }x=-1\text{ 이 구간에 없으므로 최댓값 없음}
@ 18 | \text{최댓값 }3,\ \text{최솟값 }1
f(x)=\sqrt{2x+1}\text{ 은 }[0,4]\text{ 에서 연속이고 증가}
\text{최솟값 }f(0)=1,\quad\text{최댓값 }f(4)=\sqrt9=3
@ 19 | \text{풀이 참조}
f(x)=x^3+x-1\text{ 은 }[0,1]\text{ 에서 연속}
f(0)=-1<0,\quad f(1)=1>0
@ 20 | \text{풀이 참조}
f(x)=2x^4+x^2-3x-1\text{ 은 }[-1,1]\text{ 에서 연속}
f(-1)=2+1+3-1=5>0,\quad f(1)=2+1-3-1=-1<0
