@ 01 | -2

@ 02 | -4
\frac{f(3)-f(1)}{3-1}=\frac{-9-(-1)}{2}=-4
@ 03 | -2-\Delta x
\frac{f(1+\Delta x)-f(1)}{\Delta x}=\frac{-(1+\Delta x)^2+1}{\Delta x}=\frac{-2\Delta x-(\Delta x)^2}{\Delta x}=-2-\Delta x
@ 04 | -2a-\Delta x
\frac{-(a+\Delta x)^2+a^2}{\Delta x}=\frac{-2a\Delta x-(\Delta x)^2}{\Delta x}=-2a-\Delta x

@ 06 | 5

@ 07 | 7

@ 08 | -6

@ 09 | 1

@ 10 | 1
\{(a+1)^2-1\}-(a^2-1)=2a+1=3\Rightarrow a=1
@ 11 | \frac12
\{(a+1)^2+(a+1)\}-(a^2+a)=2a+2=3\Rightarrow a=\frac12
@ 12 | -1
\{-(a+1)^2+2(a+1)\}-(-a^2+2a)=-2a+1=3\Rightarrow a=-1
@ 13 | 0

@ 14 | 2

@ 15 | 4

@ 16 | -6
