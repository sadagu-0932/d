section: 03 미분가능성과 연속성 · 04 도함수 · 05 미분법 (개념 확인 문제)
@ 25 | \text{풀이 참조}
\lim_{x\to0}|x|=0=f(0)\Rightarrow\text{연속}
\lim_{h\to0+}\frac{|h|-0}{h}=1,\quad\lim_{h\to0-}\frac{|h|-0}{h}=-1
@ 26 | \text{풀이 참조}
\lim_{x\to-1}|x+1|=0=f(-1)\Rightarrow\text{연속}
\lim_{h\to0+}\frac{|h|}{h}=1\ne\lim_{h\to0-}\frac{|h|}{h}=-1
@ 27 | f'(x)=1
f'(x)=\lim_{h\to0}\frac{(x+h+2)-(x+2)}{h}=\lim_{h\to0}\frac hh=1
@ 28 | f'(x)=2x-2
\lim_{h\to0}\frac{(x+h)^2-2(x+h)-(x^2-2x)}{h}=\lim_{h\to0}\frac{2xh+h^2-2h}{h}=\lim_{h\to0}(2x+h-2)=2x-2
@ 29 | f'(x)=8x+1
\lim_{h\to0}\frac{4(x+h)^2+(x+h)-(4x^2+x)}{h}=\lim_{h\to0}\frac{8xh+4h^2+h}{h}=8x+1
@ 30 | f'(x)=-6x+1
\lim_{h\to0}\frac{-3(x+h)^2+(x+h)+2-(-3x^2+x+2)}{h}=\lim_{h\to0}\frac{-6xh-3h^2+h}{h}=-6x+1
@ 31 | y'=1
(x)'=1
@ 32 | y'=4x^3
(x^n)'=nx^{n-1}\Rightarrow(x^4)'=4x^3
@ 33 | y'=10x^9
(x^n)'=nx^{n-1}\Rightarrow(x^{10})'=10x^{10-1}=10x^9
@ 34 | y'=0
(c)'=0\ (\text{상수함수})\Rightarrow(-1)'=0
@ 35 | y'=4x+1

@ 36 | y'=10x
y'=5\cdot2x=10x
@ 37 | y'=3x^2+1
y'=(x^3)'+(x)'-(1)'=3x^2+1-0=3x^2+1
@ 38 | y'=-4x+3
y'=-2\cdot2x+3-0=-4x+3
@ 39 | y'=4x^3+8x-1
y'=4x^3+4\cdot2x-1+0=4x^3+8x-1
@ 40 | y'=2x^3-3x^2+4x+1

@ 41 | y'=3x^2-1

@ 42 | y'=15x^4-3x^2-2

@ 43 | y'=8x+4

@ 44 | y'=4x^3-4x
y'=2(x^2-1)\cdot2x=4x^3-4x
@ 45 | y'=16x^3-15x^2+12x-5

@ 46 | y'=5x^4-4x^3-2x-1
