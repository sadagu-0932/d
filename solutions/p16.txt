section: 유형 10 함수의 극한의 대소 관계
@ 35 | ④\ 2
x>0\text{ 일 때 각 변에 }x\text{ 를 곱하면 }\frac{2x^2}{x^2+1}\le xf(x)\le\frac{2x^2+x}{x^2+1}
\lim_{x\to\infty}\frac{2x^2}{x^2+1}=2,\quad\lim_{x\to\infty}\frac{2x^2+x}{x^2+1}=2
\lim_{x\to\infty}xf(x)=2
@ 36 | 9
\lim_{x\to3}(6x-9)=9,\quad\lim_{x\to3}x^2=9
\lim_{x\to3}f(x)=9
@ 37 | ②\ 2
2(x-1)\le f(x)\le(x-1)(x+1)
x>1:\ 2\le\frac{f(x)}{x-1}\le x+1,\qquad x<1:\ x+1\le\frac{f(x)}{x-1}\le2
\lim_{x\to1}(x+1)=2\ \Rightarrow\ \lim_{x\to1}\frac{f(x)}{x-1}=2
@ 38 | ④\ 4
x\text{ 대신 }2x:\ 4x^2+1\le f(2x)\le4x^2+2
4+\frac1{x^2}\le\frac{f(2x)}{x^2}\le4+\frac2{x^2}\ \Rightarrow\ \lim_{x\to\infty}\frac{f(2x)}{x^2}=4
@ 39 | ③\ \frac14
x^2-4x+4\le f(x)\le x^2-4x+5
x\text{ 대신 }\frac x2:\ \frac{x^2}4-2x+4\le f\!\left(\frac x2\right)\le\frac{x^2}4-2x+5
\lim_{x\to\infty}\frac{\frac{x^2}4-2x+4}{x^2+1}=\lim_{x\to\infty}\frac{\frac{x^2}4-2x+5}{x^2+1}=\frac14
@ 40 | ②\ 1
x^3+2x^2-x-2=(x-1)(x+1)(x+2),\quad x^3+4x^2+x-6=(x-1)(x+2)(x+3)
x>1\text{ 이므로 }x-1\text{ 로 나누면 }(x+1)(x+2)\le f(x)\le(x+2)(x+3)
\lim_{x\to\infty}\frac{(x+1)(x+2)}{x^2+2x}=\lim_{x\to\infty}\frac{(x+2)(x+3)}{x^2+2x}=1
